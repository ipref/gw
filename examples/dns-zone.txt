; Example DNS zone entries for IPREF
; Add these records to your authoritative DNS server
;
; IMPORTANT: The hostname before the domain must match the hostname
; in your internal DNS zone for the dns-agent to create proper mappings.
;
; Format: <hostname>.<domain>.  IN  TXT  "AA <gateway_fqdn> + <reference>"

$ORIGIN example.com.
$TTL 3600

; SOA and NS records (standard DNS)
example.com.  IN  SOA  ns1.example.com. admin.example.com. (
    2024123101  ; Serial (update when making changes)
    7200        ; Refresh
    3600        ; Retry
    1209600     ; Expire
    3600        ; Minimum TTL
)

example.com.  IN  NS   ns1
example.com.  IN  NS   ns2

; Nameserver A records
ns1.example.com.  IN  A  203.0.113.10
ns2.example.com.  IN  A  203.0.113.11

; Gateway configuration
; The gateway must have an A record with its public IP
gw.example.com.       IN  A    203.0.113.5

; Published services
; Use references ≥1025 for security (makes port scanning impractical)
; The hostname (e.g., "web") must match the internal DNS zone

; Web server (matches web.internal -> 10.0.0.10)
web.example.com.       IN  TXT  "AA gw.example.com + 1025"

; SSH server (matches ssh.internal -> 10.0.0.22)
ssh.example.com.       IN  TXT  "AA gw.example.com + 1026"

; API server (matches api.internal -> 10.0.0.30)
api.example.com.       IN  TXT  "AA gw.example.com + 1027"

; Database (matches db.internal -> 10.0.0.40)
db.example.com.        IN  TXT  "AA gw.example.com + 1028"

; Alternative reference format with hyphens
; Hyphens separate groups of 16 bits
; This is useful for organizing references hierarchically
; service.example.com.   IN  TXT  "AA gw.example.com + 1-22"

; Multiple gateways example
; You can have different services on different gateways
; service2.example.com.  IN  TXT  "AA gw2.example.com + 2048"

; CNAME example (points to another IPREF-enabled host)
; www.example.com.       IN  CNAME  web.example.com.

; Note: You can also add regular A/AAAA records for services
; that should be accessible via traditional IP addressing
; traditional.example.com. IN  A  203.0.113.100
